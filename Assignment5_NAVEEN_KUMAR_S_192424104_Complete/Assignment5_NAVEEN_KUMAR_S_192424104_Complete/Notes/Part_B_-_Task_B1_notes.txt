  // Abstract payer and concrete payers (Task A3)
  #pragma once
  #include <algorithm>
  #include <map>
  #include <string>
  #include "bills.h"

  class Payer {                                                    // ABSTRACT interface
  public:
      virtual ~Payer() = default;
      virtual std::string name() const = 0;
      virtual double amountCovered(const Bill& b) const = 0;      // may throw ClaimError
  };

  class SelfPay : public Payer {
  public:
      std::string name() const override { return "Self"; }
      double amountCovered(const Bill&) const override { return 0; }   // patient pays
  everything
  };

  class Insurance : public Payer {
      std::string company; double coverLimit, coPayPercent;
  public:
      Insurance(const std::string& c, double limit, double copay)
          : company(c), coverLimit(checked(limit, "cover limit")), coPayPercent(copay)
      {
          if (copay < 0 || copay > 100) throw InvalidChargeError("co-payment must be
  0..100%");
      }
      std::string name() const override { return company; }
      double amountCovered(const Bill& b) const override
      {
          double claim = b.total() * (100 - coPayPercent) / 100;    // insurer's share
          if (claim > coverLimit)
              throw ClaimError("claim " + std::to_string(long(claim)) + " exceeds cover " +
  std::to_string(long(coverLimit)));
          return claim;
      }
  };

  class GovernmentScheme : public Payer {                          // fixed package rates
      std::map<std::string, double> rates;                         // by bill type
  public:
      GovernmentScheme() : rates{{"OP", 500}, {"IP", 30000}, {"DC", 15000}, {"ER", 5000}} {}
      std::string name() const override { return "GovtScheme"; }
      double amountCovered(const Bill& b) const override
      {
          auto it = rates.find(b.type());
          if (it == rates.end()) throw ClaimError("bill type " + b.type() + " not covered by
  the scheme");
          return std::min(it->second, b.total());                 // never more than the bill
      }
  };

  struct Settlement { double covered, patientPays; };

  inline Settlement settle(const Bill& b, const Payer& p)
  {
      double covered = p.amountCovered(b);                         // exceptions pass through
      return {covered, b.total() - covered};
  }

B1.2 – Comparison with the Part A Design (1 Mark)

The implementation follows the A1 and A3 UML exactly. Changes made while coding:
(1) a billNo was added for audit and reporting; (2) copying of Bill was deleted to
make  slicing  impossible, which  led  to  storing  bills as  std::unique_ptr<Bill> ;  (3)
validation was placed in a single helper  checked()  called from every constructor's
initialiser  list, so an invalid  bill can never be constructed;  (4)  print()  takes an
std::ostream& so it can also write to a file.

  template <class B> double sumAll(const std::vector<B>& v)      // instantiated per type, no
