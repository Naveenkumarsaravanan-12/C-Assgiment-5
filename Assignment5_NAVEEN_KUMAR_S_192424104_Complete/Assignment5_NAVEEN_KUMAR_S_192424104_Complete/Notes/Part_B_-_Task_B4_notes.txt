B4.1 – A Practical Extension (2 Marks)

A new bill type ( TeleconsultBill ) and a new payer ( CorporateTieUp ) were added in a new
header; no existing file was modified:

  B4/extensions.h

  // B4.1 - New bill type and new payer, added in a NEW file. No existing class is modified.
  #pragma once
  #include "../common/payers.h"

  class TeleconsultBill : public Bill {               // new bill type
      int minutes; double perMinute, platformFee;
  public:
      TeleconsultBill(const std::string& p, int no, int mins, double rate, double fee)
          : Bill(p, no), minutes(int(checked(mins, "minutes"))), perMinute(checked(rate, "per-
  minute rate")),
            platformFee(checked(fee, "platform fee")) {}
      double total() const override { return minutes * perMinute + platformFee; }
      std::string type() const override { return "TC"; }
  };

  class CorporateTieUp : public Payer {               // new payer
      std::string company; double percent, perBillCap;
  public:
      CorporateTieUp(const std::string& c, double pct, double cap) : company(c),
  percent(pct), perBillCap(cap) {}
      std::string name() const override { return company; }
      double amountCovered(const Bill& b) const override
      {
          if (b.type() == "ER") throw ClaimError(company + " tie-up excludes emergency
  bills");
          double share = b.total() * percent / 100;
          return share < perBillCap ? share : perBillCap;
      }
  };

  // B4 - Regression tests (all B1 results), tests of the extension, and error paths under
  ASan
  #include <cmath>
  #include <functional>
  #include <memory>
  #include <vector>
  #include "extensions.h"

  int passed = 0, failed = 0;
  void expect(const std::string& what, double got, double want)
  {
      bool ok = std::fabs(got - want) < 1e-6;
      (ok ? passed : failed)++;
      std::cout << (ok ? "  PASS  " : "  FAIL  ") << std::left << std::setw(42) << what <<
  std::right
                << std::setw(10) << got << " (expected " << want << ")\n";
  }
  void expectThrow(const std::string& what, const std::function<void()>& f, const
  std::string& kind)
  {
      try { f(); ++failed; std::cout << "  FAIL  " << what << ": no exception\n"; }
      catch (const BillingError& e) { ++passed; std::cout << "  PASS  " << std::left <<
  std::setw(42) << what << std::right << " threw " << kind << ": " << e.what() << '\n'; }
  }

  // A5.1 design: settle() logs and RETHROWS; the day loop reports and continues
  double settleLogged(const Bill& b, const Payer& p)
  {
      try { return settle(b, p).covered; }
      catch (const ClaimError& e) { std::cout << "    [audit log] bill " << b.name() << ": "
  << e.what() << '\n'; throw; }
  }

  int main()
  {
      std::cout << std::fixed << std::setprecision(2);
      std::cout << "Earlier tests (B1.1 table):\n";
      OutPatientBill op("Ravi", 101, 500, 1200);
      InPatientBill ip("Meena", 102, 3, 2500, 40000, 6500);
      DayCareBill dc("Kumar", 103, "CATARACT");
      EmergencyBill er("Latha", 104, 2000, 5, 800);
      SelfPay self; Insurance ins("StarHealth", 60000, 10); GovernmentScheme gov;
      expect("OP total", op.total(), 1700);         expect("IP total", ip.total(), 54000);
      expect("DC total", dc.total(), 25000);        expect("ER total", er.total(), 6000);
      expect("Self covers OP", settle(op, self).covered, 0);
      expect("Insurance covers IP (90%)", settle(ip, ins).covered, 48600);
      expect("Govt covers DC (package 15000)", settle(dc, gov).covered, 15000);
      expect("Govt covers ER (package 5000)", settle(er, gov).covered, 5000);

      std::cout << "Extension (new bill type TC, new payer CorporateTieUp):\n";
      TeleconsultBill tc("Nisha", 105, 20, 30, 99);
      CorporateTieUp corp("InfoTech Ltd", 80, 20000);
      expect("TC total 20 min x 30 + 99", tc.total(), 699);
      expect("Corporate covers TC (80%)", settle(tc, corp).covered, 559.2);
      expect("Corporate covers IP (cap 20000)", settle(ip, corp).covered, 20000);
      expect("Insurance covers TC (90%)", settle(tc, ins).covered, 629.1);
      expectThrow("Govt scheme on TC (not in its table)", [&] { settle(tc, gov); },
  "ClaimError");
      expectThrow("Corporate on ER (excluded)", [&] { settle(er, corp); }, "ClaimError");

      std::cout << "Error paths (exceptions thrown and caught; objects must be released):\n";
      expectThrow("negative consultation", [] { OutPatientBill b("X", 1, -1, 0); },
  "InvalidChargeError");
      expectThrow("unknown package", [] { auto b = std::make_unique<DayCareBill>("Y", 2,
  "LASIK"); }, "UnknownPackageError");
      expectThrow("negative tele minutes", [] { TeleconsultBill b("Z", 3, -5, 30, 99); },
  "InvalidChargeError");

      std::cout << "Processing a day's bills through Bill* (logged and rethrown claim errors):
  \n";
      std::vector<std::unique_ptr<Bill>> day;
      day.push_back(std::make_unique<InPatientBill>("Arun", 201, 12, 3000, 30000,
  4000));     // 70000 -> 63000 > cover
      day.push_back(std::make_unique<TeleconsultBill>("Bala", 202, 15, 30, 99));
      day.push_back(std::make_unique<OutPatientBill>("Chitra", 203, 500, 800));
      double covered = 0; int rejected = 0;
      for (const auto& b : day) {
          try { covered += settleLogged(*b, ins); }
          catch (const BillingError&) { ++rejected; std::cout << "    [report] " << b->type()
  << " bill of " << b->name() << " rejected - kept for manual review\n"; }
      }
      expect("insurer total for the day (494.10 + 1170)", covered, 494.1 + 1170);
      std::cout << "Result: " << passed << " passed, " << failed << " failed; " << rejected
  << " bill(s) sent to manual review, none lost\n";
      return failed;
  }

  ALGORITHM TotalFromRules(bill, rules)
  BEGIN
    rule ← rules.Lookup(bill.type)                 // hash lookup, O(1) average
    IF rule = NULL THEN THROW UnknownPackageError END IF
    RETURN Evaluate(rule.expression, bill.fields)  // interpret the expression: O(size
  of rule)
  END

 Overall Conclusion

The  hospital  billing  system  demonstrates  the  two  pillars  of  CO5.  Runtime
polymorphism – abstract Bill and Payer classes, virtual and pure virtual functions,
virtual destructors and late binding through vtables – lets the program settle every
combination of bill and payer through one interface and accept new bill types and
payers without modifying tested code. Exception handling – a hierarchy of error
classes, throwing from constructors and business rules, catching at the right level,
logging and rethrowing with  throw; , and RAII for automatic cleanup during stack
unwinding – ensures that invalid bills are rejected with clear reasons and that no bill or
resource is lost. The measurements show that the costs of both mechanisms are small
and predictable at the hospital group's scale, which supports the final decision to adopt
the polymorphic, exception-based design.
